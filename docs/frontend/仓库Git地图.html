07复现这张地图

# 主干：提交图与分支
git log --all --graph --oneline --decorate

# 逐次提交的改动规模
git log --reverse --pretty=format:"%h %ad %s" --date=short --shortstat

# 每次提交动了哪些目录（地图里“涉及目录”一列）
git log --reverse --pretty=format:"%h" --name-only | head -40

# 类型分布
git log --pretty=format:"%s" | sed 's/(.*//' | cut -d: -f1 | sort | uniq -c | sort -rn

# 累计变更行
git log --numstat --pretty=format:"" | awk 'NF==3{a+=$1;d+=$2} END{print "+" a, "-" d}'
